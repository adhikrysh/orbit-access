cmake_minimum_required(VERSION 3.20)
project(orbit_access VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
add_compile_options(-Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)

option(ORBIT_SANITIZERS "Enable address and undefined-behavior checks" OFF)
add_library(orbit src/dynamics.cpp src/access.cpp)
target_include_directories(orbit PUBLIC include)
target_compile_features(orbit PUBLIC cxx_std_20)
if(ORBIT_SANITIZERS)
  target_compile_options(orbit PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(orbit PUBLIC -fsanitize=address,undefined)
endif()
add_executable(orbit-access src/main.cpp)
target_link_libraries(orbit-access PRIVATE orbit)
enable_testing()
add_executable(orbit-tests tests/test_orbit.cpp)
target_link_libraries(orbit-tests PRIVATE orbit)
add_test(NAME physics-and-access COMMAND orbit-tests)
